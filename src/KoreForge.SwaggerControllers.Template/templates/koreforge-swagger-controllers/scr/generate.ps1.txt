<#
.SYNOPSIS
  Stage-2 generator for KoreForge.SwaggerControllers consumer repos.

.DESCRIPTION
  Reads every swaggers/<ApiName>/v<N>/metadata.json file under the repo and
  (re)emits the strongly-typed Generated/ tree for each API library under src/.
  Scaffolded files (decorators, ServiceCollectionExtensions, .csproj) are emitted
  once and never overwritten. Run this after adding or updating metadata.json files.

.NOTES
  Requires PowerShell 7+. No external modules needed.
  metadata.json must follow the KoreForge.SwaggerControllers metadata schema.
#>

param(
    [string]$RepoRoot = (Split-Path $PSScriptRoot -Parent)
)

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

$swaggersDir = Join-Path $RepoRoot 'swaggers'
$srcDir      = Join-Path $RepoRoot 'src'
$tstDir      = Join-Path $RepoRoot 'tst'

# ─────────────────────────────────────────────
# Helper: Write-FileIfChanged — avoids unnecessary writes
# ─────────────────────────────────────────────
function Write-FileIfChanged {
    param([string]$Path, [string]$Content)
    $dir = Split-Path $Path -Parent
    if (-not (Test-Path $dir)) { New-Item -ItemType Directory -Path $dir -Force | Out-Null }
    $Content = $Content -replace "`r`n", "`n"
    if (Test-Path $Path) {
        $existing = (Get-Content $Path -Raw -ErrorAction SilentlyContinue) -replace "`r`n", "`n"
        if ($existing -eq $Content) { return }
    }
    [System.IO.File]::WriteAllText($Path, $Content, [System.Text.Encoding]::UTF8)
    Write-Host "  wrote: $Path"
}

# ─────────────────────────────────────────────
# Helper: Build Refit attribute for an operation
# ─────────────────────────────────────────────
function Get-RefitAttribute($Op) {
    $method = $Op.httpMethod
    $path   = $Op.pathTemplate
    switch ($method.ToUpper()) {
        'GET'    { return "[Get(`"$path`")]" }
        'POST'   { return "[Post(`"$path`")]" }
        'PUT'    { return "[Put(`"$path`")]" }
        'DELETE' { return "[Delete(`"$path`")]" }
        'PATCH'  { return "[Patch(`"$path`")]" }
        default  { return "[Get(`"$path`")]" }
    }
}

# ─────────────────────────────────────────────
# Helper: Build method signature parameters
# ─────────────────────────────────────────────
function Get-ServiceMethodParams($Op) {
    $parts = @()
    foreach ($p in $Op.parameters) {
        if ($p.in -eq 'header') { continue }
        if ($p.isBodyParam) {
            $parts += "$($p.csharpType) request"
        } else {
            $camel = $p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1)
            $parts += "$($p.csharpType) $camel"
        }
    }
    $parts += "CancellationToken cancellationToken = default"
    return $parts -join ', '
}

function Get-ServiceCallArgs($Op) {
    $parts = @()
    foreach ($p in $Op.parameters) {
        if ($p.in -eq 'header') { continue }
        if ($p.isBodyParam) {
            $parts += "request"
        } else {
            $camel = $p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1)
            $parts += $camel
        }
    }
    $parts += "cancellationToken"
    return $parts -join ', '
}

function Get-RefitCallArgs($Op) {
    $parts = @()
    foreach ($p in $Op.parameters) {
        if ($p.isBodyParam) {
            $parts += "request"
        } elseif ($p.in -eq 'header') {
            continue
        } elseif ($p.in -eq 'path') {
            $parts += $p.name
        } else {
            $camel = $p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1)
            $parts += $camel
        }
    }
    $parts += "cancellationToken"
    return $parts -join ', '
}

function Get-RefitMethodParamsNoHeader($Op) {
    $parts = @()
    foreach ($p in $Op.parameters) {
        if ($p.in -eq 'header') { continue }
        if ($p.isBodyParam) {
            $parts += "[Body] $($p.csharpType) request"
        } elseif ($p.in -eq 'query') {
            $parts += "[Query] $($p.csharpType) $($p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1))"
        } elseif ($p.in -eq 'path') {
            $parts += "$($p.csharpType) $($p.name)"
        }
    }
    $parts += "CancellationToken cancellationToken = default"
    return $parts -join ', '
}

function Get-ResponseType($Op) {
    if ($Op.successResponseDtoRef) {
        return $Op.successResponseDtoRef
    }
    return 'object'
}

# ─────────────────────────────────────────────
# Collect all metadata.json files
# ─────────────────────────────────────────────
$metadataFiles = @(Get-ChildItem -Path $swaggersDir -Filter 'metadata.json' -Recurse | Sort-Object FullName)
if ($metadataFiles.Count -eq 0) {
    Write-Error "No metadata.json files found under $swaggersDir"
    return
}

Write-Host "Found $($metadataFiles.Count) metadata.json files."
Write-Host ""

# Determine test project name from .slnx
$slnxFile = Get-ChildItem -Path $RepoRoot -Filter '*.slnx' -File | Select-Object -First 1
$testProjName = if ($slnxFile) { "$($slnxFile.BaseName).Tests" } else { "Tests" }

$projectRefs = @()

foreach ($mf in $metadataFiles) {
    $meta     = Get-Content $mf.FullName -Raw | ConvertFrom-Json
    $ns       = $meta.namespace
    $apiName  = $meta.apiName
    $version  = $meta.folderVersion
    $kebab    = $meta.kebabName
    $projName = $meta.projectName
    $basePath = $meta.basePath

    Write-Host "=== Generating $projName ==="

    $projDir = Join-Path $srcDir $projName
    $genDir  = Join-Path $projDir 'Generated'
    $dtosDir = Join-Path $genDir 'Dtos'

    if (Test-Path $genDir) { Remove-Item $genDir -Recurse -Force }
    New-Item -ItemType Directory -Path $dtosDir -Force | Out-Null

    # ───────────────────────────────────────
    # 1. Generate DTOs
    # ───────────────────────────────────────
    foreach ($dto in $meta.dtos) {
        $dtoName = $dto.csharpName
        $sb = [System.Text.StringBuilder]::new()
        [void]$sb.AppendLine("// <auto-generated />")
        [void]$sb.AppendLine('#nullable enable')
        [void]$sb.AppendLine("using System.Text.Json.Serialization;")
        [void]$sb.AppendLine("")
        [void]$sb.AppendLine("namespace $ns;")
        [void]$sb.AppendLine("")
        [void]$sb.AppendLine("public sealed class $dtoName")
        [void]$sb.AppendLine("{")
        foreach ($prop in $dto.properties) {
            $propType = $prop.csharpType
            $isRequired = $null -ne ($prop.PSObject.Properties['required']) -and $prop.required -eq $true
            # Non-required properties that aren't already nullable get ? suffix
            if (-not $isRequired -and -not $propType.EndsWith('?')) {
                $propType = "$($propType)?"
            }
            $requiredKeyword = if ($isRequired) { "required " } else { "" }
            [void]$sb.AppendLine("    [JsonPropertyName(`"$($prop.jsonName)`")]")
            [void]$sb.AppendLine("    public ${requiredKeyword}${propType} $($prop.csharpName) { get; set; }")
            [void]$sb.AppendLine("")
        }
        [void]$sb.AppendLine("}")
        Write-FileIfChanged -Path (Join-Path $dtosDir "$dtoName.g.cs") -Content $sb.ToString()
    }

    # ───────────────────────────────────────
    # 2. Generate Refit interface
    # ───────────────────────────────────────
    $refitSb = [System.Text.StringBuilder]::new()
    [void]$refitSb.AppendLine("// <auto-generated />")
    [void]$refitSb.AppendLine('#nullable enable')
    [void]$refitSb.AppendLine("using Refit;")
    [void]$refitSb.AppendLine("")
    [void]$refitSb.AppendLine("namespace $ns;")
    [void]$refitSb.AppendLine("")
    [void]$refitSb.AppendLine("public interface I${apiName}ExternalClient")
    [void]$refitSb.AppendLine("{")
    foreach ($op in $meta.operations) {
        [void]$refitSb.AppendLine("    $(Get-RefitAttribute $op)")
        [void]$refitSb.AppendLine("    Task<$(Get-ResponseType $op)> $($op.refitMethodName)($(Get-RefitMethodParamsNoHeader $op));")
        [void]$refitSb.AppendLine("")
    }
    [void]$refitSb.AppendLine("}")
    Write-FileIfChanged -Path (Join-Path $genDir "I${apiName}ExternalClient.g.cs") -Content $refitSb.ToString()

    # ───────────────────────────────────────
    # 3. Generate Service interface
    # ───────────────────────────────────────
    $svcIfaceSb = [System.Text.StringBuilder]::new()
    [void]$svcIfaceSb.AppendLine("// <auto-generated />")
    [void]$svcIfaceSb.AppendLine('#nullable enable')
    [void]$svcIfaceSb.AppendLine("using KoreForge.SwaggerControllers;")
    [void]$svcIfaceSb.AppendLine("")
    [void]$svcIfaceSb.AppendLine("namespace $ns;")
    [void]$svcIfaceSb.AppendLine("")
    [void]$svcIfaceSb.AppendLine("public interface I${apiName}Service")
    [void]$svcIfaceSb.AppendLine("{")
    foreach ($op in $meta.operations) {
        [void]$svcIfaceSb.AppendLine("    Task<Outcome<$(Get-ResponseType $op)>> $($op.internalMethodName)($(Get-ServiceMethodParams $op));")
        [void]$svcIfaceSb.AppendLine("")
    }
    [void]$svcIfaceSb.AppendLine("}")
    Write-FileIfChanged -Path (Join-Path $genDir "I${apiName}Service.g.cs") -Content $svcIfaceSb.ToString()

    # ───────────────────────────────────────
    # 4. Generate Core Service
    # ───────────────────────────────────────
    $coreSb = [System.Text.StringBuilder]::new()
    [void]$coreSb.AppendLine("// <auto-generated />")
    [void]$coreSb.AppendLine('#nullable enable')
    [void]$coreSb.AppendLine("using KoreForge.SwaggerControllers;")
    [void]$coreSb.AppendLine("using Microsoft.Extensions.Logging;")
    [void]$coreSb.AppendLine("")
    [void]$coreSb.AppendLine("namespace $ns;")
    [void]$coreSb.AppendLine("")
    [void]$coreSb.AppendLine("internal sealed class ${apiName}Service : I${apiName}Service")
    [void]$coreSb.AppendLine("{")
    [void]$coreSb.AppendLine('    private const string ClassName = "' + $apiName + 'Service";')
    [void]$coreSb.AppendLine("    private readonly I${apiName}ExternalClient _client;")
    [void]$coreSb.AppendLine("    private readonly ILogger<${apiName}Service> _logger;")
    [void]$coreSb.AppendLine("")
    [void]$coreSb.AppendLine("    public ${apiName}Service(I${apiName}ExternalClient client, ILogger<${apiName}Service> logger)")
    [void]$coreSb.AppendLine("    {")
    [void]$coreSb.AppendLine("        _client = client;")
    [void]$coreSb.AppendLine("        _logger = logger;")
    [void]$coreSb.AppendLine("    }")
    foreach ($op in $meta.operations) {
        $respType   = Get-ResponseType $op
        $methodName = $op.internalMethodName
        $paramStr   = Get-ServiceMethodParams $op
        $callArgs   = Get-RefitCallArgs $op
        $opKey      = $op.operationKey

        [void]$coreSb.AppendLine("")
        [void]$coreSb.AppendLine("    public async Task<Outcome<$respType>> $methodName($paramStr)")
        [void]$coreSb.AppendLine("    {")
        [void]$coreSb.AppendLine("        var correlationId = Guid.NewGuid().ToString();")
        [void]$coreSb.AppendLine("        try")
        [void]$coreSb.AppendLine("        {")
        [void]$coreSb.AppendLine("            var result = await _client.$($op.refitMethodName)($callArgs);")
        [void]$coreSb.AppendLine("            return Outcome<$respType>.Ok(result, correlationId);")
        [void]$coreSb.AppendLine("        }")
        [void]$coreSb.AppendLine("        catch (Refit.ApiException ex)")
        [void]$coreSb.AppendLine("        {")
        [void]$coreSb.AppendLine('            _logger.LogExceptionStructured(ex, ClassName, "' + $methodName + '", "' + $opKey + '", correlationId,')
        [void]$coreSb.AppendLine('                new Dictionary<string, object?>')
        [void]$coreSb.AppendLine('                {')
        [void]$coreSb.AppendLine('                    ["HttpStatusCode"] = (int)ex.StatusCode,')
        [void]$coreSb.AppendLine('                    ["ReasonPhrase"] = ex.ReasonPhrase,')
        [void]$coreSb.AppendLine('                    ["RequestUri"] = ex.Uri?.ToString()')
        [void]$coreSb.AppendLine('                });')
        [void]$coreSb.AppendLine("            var error = new ErrorInfo")
        [void]$coreSb.AppendLine("            {")
        [void]$coreSb.AppendLine("                Origin = ErrorOrigin.External,")
        [void]$coreSb.AppendLine("                Kind = ex.StatusCode == System.Net.HttpStatusCode.Unauthorized || ex.StatusCode == System.Net.HttpStatusCode.Forbidden")
        [void]$coreSb.AppendLine("                    ? ErrorKind.Authorization")
        [void]$coreSb.AppendLine("                    : ErrorKind.Technical,")
        [void]$coreSb.AppendLine("                Code = `"$($opKey)_FAILED`",")
        [void]$coreSb.AppendLine("                Message = ex.Message,")
        [void]$coreSb.AppendLine("                ExternalDetail = new ExternalErrorDetail")
        [void]$coreSb.AppendLine("                {")
        [void]$coreSb.AppendLine("                    DependencyName = `"$apiName`",")
        [void]$coreSb.AppendLine("                    HttpStatusCode = (int)ex.StatusCode,")
        [void]$coreSb.AppendLine("                    RawResponse = ex.Content")
        [void]$coreSb.AppendLine("                }")
        [void]$coreSb.AppendLine("            };")
        [void]$coreSb.AppendLine("            return Outcome<$respType>.Fail(error, correlationId);")
        [void]$coreSb.AppendLine("        }")
        [void]$coreSb.AppendLine("        catch (Exception ex)")
        [void]$coreSb.AppendLine("        {")
        [void]$coreSb.AppendLine('            _logger.LogExceptionStructured(ex, ClassName, "' + $methodName + '", "' + $opKey + '", correlationId);')
        [void]$coreSb.AppendLine("            var error = new ErrorInfo")
        [void]$coreSb.AppendLine("            {")
        [void]$coreSb.AppendLine("                Origin = ErrorOrigin.Internal,")
        [void]$coreSb.AppendLine("                Kind = ErrorKind.Technical,")
        [void]$coreSb.AppendLine("                Code = `"$($opKey)_INTERNAL_ERROR`",")
        [void]$coreSb.AppendLine("                Message = ex.Message")
        [void]$coreSb.AppendLine("            };")
        [void]$coreSb.AppendLine("            return Outcome<$respType>.Fail(error, correlationId);")
        [void]$coreSb.AppendLine("        }")
        [void]$coreSb.AppendLine("    }")
    }
    [void]$coreSb.AppendLine("}")
    Write-FileIfChanged -Path (Join-Path $genDir "${apiName}Service.g.cs") -Content $coreSb.ToString()

    # ───────────────────────────────────────
    # 5. Generate Logging Decorator
    # ───────────────────────────────────────
    $logSb = [System.Text.StringBuilder]::new()
    [void]$logSb.AppendLine("// <auto-generated />")
    [void]$logSb.AppendLine('#nullable enable')
    [void]$logSb.AppendLine("using KoreForge.SwaggerControllers;")
    [void]$logSb.AppendLine("using Microsoft.Extensions.Logging;")
    [void]$logSb.AppendLine("")
    [void]$logSb.AppendLine("namespace $ns;")
    [void]$logSb.AppendLine("")
    [void]$logSb.AppendLine("internal sealed class ${apiName}LoggingDecorator : I${apiName}Service")
    [void]$logSb.AppendLine("{")
    [void]$logSb.AppendLine('    private const string ClassName = "' + $apiName + 'LoggingDecorator";')
    [void]$logSb.AppendLine("    private readonly I${apiName}Service _inner;")
    [void]$logSb.AppendLine("    private readonly ILogger<${apiName}LoggingDecorator> _logger;")
    [void]$logSb.AppendLine("")
    [void]$logSb.AppendLine("    public ${apiName}LoggingDecorator(I${apiName}Service inner, ILogger<${apiName}LoggingDecorator> logger)")
    [void]$logSb.AppendLine("    {")
    [void]$logSb.AppendLine("        _inner = inner;")
    [void]$logSb.AppendLine("        _logger = logger;")
    [void]$logSb.AppendLine("    }")
    foreach ($op in $meta.operations) {
        $respType   = Get-ResponseType $op
        $methodName = $op.internalMethodName
        $paramStr   = Get-ServiceMethodParams $op
        $callArgs   = Get-ServiceCallArgs $op
        $opKey      = $op.operationKey

        $scopeEntries = @()
        $scopeEntries += '            ["ClassName"] = ClassName'
        $scopeEntries += '            ["MethodName"] = "' + $methodName + '"'
        $scopeEntries += '            ["OperationKey"] = "' + $opKey + '"'
        foreach ($p in $op.parameters) {
            if ($p.in -eq 'header') { continue }
            if ($p.isBodyParam) {
                $scopeEntries += '            ["RequestType"] = request.GetType().Name'
                $scopeEntries += '            ["HasRequestBody"] = true'
            } else {
                $camel = $p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1)
                $scopeEntries += '            ["' + $p.csharpName + '"] = ' + $camel
            }
        }

        $logParamTokens = @()
        $logParamArgs   = @()
        foreach ($p in $op.parameters) {
            if ($p.in -eq 'header') { continue }
            if ($p.isBodyParam) { continue }
            $camel = $p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1)
            $logParamTokens += "{$($p.csharpName)}"
            $logParamArgs   += $camel
        }
        $paramLogTemplate = ''
        $paramLogArgs     = ''
        if ($logParamTokens.Count -gt 0) {
            $paramLogTemplate = ' with ' + ($logParamTokens -join ', ')
            $paramLogArgs     = ', ' + ($logParamArgs -join ', ')
        }

        [void]$logSb.AppendLine("")
        [void]$logSb.AppendLine("    public async Task<Outcome<$respType>> $methodName($paramStr)")
        [void]$logSb.AppendLine("    {")
        [void]$logSb.AppendLine('        using var scope = _logger.BeginScope(new Dictionary<string, object?>')
        [void]$logSb.AppendLine('        {')
        foreach ($entry in $scopeEntries) {
            [void]$logSb.AppendLine("$entry,")
        }
        [void]$logSb.AppendLine('        });')
        [void]$logSb.AppendLine("")
        [void]$logSb.AppendLine('        _logger.LogInformation(')
        [void]$logSb.AppendLine('            "{ClassName}.{MethodName} invoking {OperationKey}' + $paramLogTemplate + '",')
        [void]$logSb.AppendLine('            ClassName, "' + $methodName + '", "' + $opKey + '"' + $paramLogArgs + ');')
        [void]$logSb.AppendLine("")
        [void]$logSb.AppendLine("        var result = await _inner.$methodName($callArgs);")
        [void]$logSb.AppendLine("")
        [void]$logSb.AppendLine('        _logger.LogOutcomeStructured(result, ClassName, "' + $methodName + '", "' + $opKey + '");')
        [void]$logSb.AppendLine("")
        [void]$logSb.AppendLine("        return result;")
        [void]$logSb.AppendLine("    }")
    }
    [void]$logSb.AppendLine("}")
    Write-FileIfChanged -Path (Join-Path $genDir "${apiName}LoggingDecorator.g.cs") -Content $logSb.ToString()

    # ───────────────────────────────────────
    # 6. Generate Controller
    # ───────────────────────────────────────
    $ctrlSb = [System.Text.StringBuilder]::new()
    [void]$ctrlSb.AppendLine("// <auto-generated />")
    [void]$ctrlSb.AppendLine('#nullable enable')
    [void]$ctrlSb.AppendLine("using KoreForge.SwaggerControllers;")
    [void]$ctrlSb.AppendLine('using Microsoft.AspNetCore.Authorization;')
    [void]$ctrlSb.AppendLine("using Microsoft.AspNetCore.Mvc;")
    [void]$ctrlSb.AppendLine("")
    [void]$ctrlSb.AppendLine("namespace $ns;")
    [void]$ctrlSb.AppendLine("")
    [void]$ctrlSb.AppendLine("[ApiController]")
    [void]$ctrlSb.AppendLine("[Route(`"api/$kebab/v$version`")]")
    [void]$ctrlSb.AppendLine("public sealed class ${apiName}Controller : ControllerBase")
    [void]$ctrlSb.AppendLine("{")
    [void]$ctrlSb.AppendLine("    private readonly I${apiName}Service _service;")
    [void]$ctrlSb.AppendLine("")
    [void]$ctrlSb.AppendLine("    public ${apiName}Controller(I${apiName}Service service)")
    [void]$ctrlSb.AppendLine("    {")
    [void]$ctrlSb.AppendLine("        _service = service;")
    [void]$ctrlSb.AppendLine("    }")
    foreach ($op in $meta.operations) {
        $respType   = Get-ResponseType $op
        $methodName = $op.internalMethodName
        $ctrlRoute  = $op.controllerRoute
        $httpMethod = $op.httpMethod.ToUpper()
        $callArgs   = Get-ServiceCallArgs $op
        $permName   = $methodName -replace 'Async$', ''

        $actionParams = @()
        foreach ($p in $op.parameters) {
            if ($p.in -eq 'header') { continue }
            if ($p.isBodyParam) {
                $actionParams += "[FromBody] $($p.csharpType) request"
            } elseif ($p.in -eq 'path') {
                $actionParams += "[FromRoute] $($p.csharpType) $($p.name)"
            } elseif ($p.in -eq 'query') {
                $camel = $p.csharpName.Substring(0,1).ToLower() + $p.csharpName.Substring(1)
                $actionParams += "[FromQuery] $($p.csharpType) $camel"
            }
        }
        $actionParams += "CancellationToken cancellationToken"
        $actionParamStr = $actionParams -join ', '

        $httpAttr = switch ($httpMethod) {
            'GET'    { "HttpGet" }; 'POST'   { "HttpPost" }; 'PUT'    { "HttpPut" }
            'DELETE' { "HttpDelete" }; 'PATCH'  { "HttpPatch" }; default  { "HttpGet" }
        }

        [void]$ctrlSb.AppendLine("")
        [void]$ctrlSb.AppendLine("    [Authorize(Policy = Permissions.$permName)]")
        [void]$ctrlSb.AppendLine("    [$httpAttr(`"$ctrlRoute`")]")
        [void]$ctrlSb.AppendLine("    public async Task<IActionResult> $methodName($actionParamStr)")
        [void]$ctrlSb.AppendLine("    {")
        [void]$ctrlSb.AppendLine("        var outcome = await _service.$methodName($callArgs);")
        [void]$ctrlSb.AppendLine("        return outcome.ToActionResult();")
        [void]$ctrlSb.AppendLine("    }")
    }
    [void]$ctrlSb.AppendLine("}")
    Write-FileIfChanged -Path (Join-Path $genDir "${apiName}Controller.g.cs") -Content $ctrlSb.ToString()

    # ───────────────────────────────────────
    # 7. Generate Permissions constants
    # ───────────────────────────────────────
    $permSb = [System.Text.StringBuilder]::new()
    [void]$permSb.AppendLine("// <auto-generated />")
    [void]$permSb.AppendLine('#nullable enable')
    [void]$permSb.AppendLine("")
    [void]$permSb.AppendLine("namespace $ns;")
    [void]$permSb.AppendLine("")
    [void]$permSb.AppendLine("/// <summary>")
    [void]$permSb.AppendLine("/// Compile-time permission policy name constants for this API.")
    [void]$permSb.AppendLine("/// Register these policies at startup to define authorization requirements.")
    [void]$permSb.AppendLine("/// </summary>")
    [void]$permSb.AppendLine("public static class Permissions")
    [void]$permSb.AppendLine("{")
    $allPermNames = @()
    foreach ($op in $meta.operations) {
        $permName   = $op.internalMethodName -replace 'Async$', ''
        $policyName = "$apiName.V$version.$permName"
        [void]$permSb.AppendLine("    public const string $permName = `"$policyName`";")
        $allPermNames += $permName
    }
    [void]$permSb.AppendLine("")
    [void]$permSb.AppendLine("    /// <summary>All permission policy names for this API.</summary>")
    [void]$permSb.AppendLine("    public static readonly string[] All = [$($allPermNames -join ', ')];")
    [void]$permSb.AppendLine("}")
    Write-FileIfChanged -Path (Join-Path $genDir 'Permissions.g.cs') -Content $permSb.ToString()

    # ───────────────────────────────────────
    # 8. Scaffold ServiceCollectionExtensions (once)
    # ───────────────────────────────────────
    $diPath = Join-Path $projDir "ServiceCollectionExtensions.cs"
    if (-not (Test-Path $diPath)) {
        $diSb = [System.Text.StringBuilder]::new()
        [void]$diSb.AppendLine("using KoreForge.SwaggerControllers;")
        [void]$diSb.AppendLine("using Microsoft.AspNetCore.Mvc.ApplicationParts;")
        [void]$diSb.AppendLine("using Microsoft.Extensions.DependencyInjection;")
        [void]$diSb.AppendLine("using Microsoft.Extensions.Logging;")
        [void]$diSb.AppendLine("using Refit;")
        [void]$diSb.AppendLine("")
        [void]$diSb.AppendLine("namespace $ns;")
        [void]$diSb.AppendLine("")
        [void]$diSb.AppendLine("public static class ServiceCollectionExtensions")
        [void]$diSb.AppendLine("{")
        [void]$diSb.AppendLine("    /// <summary>")
        [void]$diSb.AppendLine("    /// Registers the $apiName V$version Refit client, auth handler, and decorator chain.")
        [void]$diSb.AppendLine("    /// Scaffolded once — edit this file to adjust the decorator chain or base URL.")
        [void]$diSb.AppendLine("    /// </summary>")
        [void]$diSb.AppendLine("    public static IServiceCollection Add${apiName}V${version}Services(")
        [void]$diSb.AppendLine("        this IServiceCollection services,")
        [void]$diSb.AppendLine("        Action<ExternalAuthOptions> configureAuth,")
